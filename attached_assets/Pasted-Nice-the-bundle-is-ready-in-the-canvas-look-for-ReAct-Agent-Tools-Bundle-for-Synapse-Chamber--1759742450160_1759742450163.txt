# react_agent_integrated.py
import json, time, logging
from typing import Any, Dict, List

logger = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO)

# imports that should exist in your project
from ai_controller import AiController
from memory_system import MemorySystem
from training_engine import TrainingEngine
from browser_automation import BrowserController

# minimal local tool wrappers (adjust if your method names differ)
ai = AiController()
memory = MemorySystem()
training = TrainingEngine()
browser = BrowserController()

# Define the tool registry used by the agent
def _wrap_call_ai(payload):
    return ai.call_platform(payload.get("platform"), payload.get("prompt"), **(payload.get("opts", {})))

def _wrap_search_web(payload):
    return ai.web_search(payload.get("query")) if hasattr(ai, "web_search") else {"error":"no web_search"}

def _wrap_memory_retrieve(payload):
    return memory.retrieve(payload.get("query"), top_k=payload.get("top_k",5))

def _wrap_memory_store(payload):
    return memory.store(payload.get("record"))

def _wrap_browse_page(payload):
    return browser.navigate_and_extract(payload.get("url"), payload.get("selectors"))

def _wrap_fill_form(payload):
    return browser.fill_form(payload.get("url"), payload.get("data"))

def _wrap_solve_captcha(payload):
    return browser.solve_captcha(payload.get("screenshot_path"))

def _wrap_run_training_cycle(payload):
    return training.run_cycle(payload)

def _wrap_enqueue_training(payload):
    return training.enqueue_training_task(payload.get("task"))

TOOLS = {
    "call_ai": _wrap_call_ai,
    "search_web": _wrap_search_web,
    "memory_retrieve": _wrap_memory_retrieve,
    "memory_store": _wrap_memory_store,
    "browse_page": _wrap_browse_page,
    "fill_form": _wrap_fill_form,
    "solve_captcha": _wrap_solve_captcha,
    "run_training_cycle": _wrap_run_training_cycle,
    "enqueue_training": _wrap_enqueue_training,
}

class ReActAgentIntegrated:
    def __init__(self, max_steps:int=6):
        self.ai = ai
        self.memory = memory
        self.training = training
        self.browser = browser
        self.max_steps = max_steps
        self.approval_required = {"create_pr","delete_resource","transfer_funds"}
        self.audit_log_path = "audit.log"

    def _write_audit(self, obj:Dict[str,Any]):
        with open(self.audit_log_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(obj, default=str) + "\n")

    def _build_prompt(self, task:str, context_snippets:List[str], history:List[Dict[str,Any]]) -> str:
        tools_list = ", ".join(sorted(TOOLS.keys()))
        ctx = "\n".join(f"- {s}" for s in (context_snippets or [])[:5])
        history_snip = json.dumps(history[-6:], default=str)
        return (
            "You are AION — an autonomous agent. RETURN A SINGLE JSON OBJECT ONLY. NO CHAIN-OF-THOUGHT.\n"
            f"Keys: action (one of: {tools_list} or 'finish'), action_input (object). Optional: comment.\n"
            f"Task: {task}\nContext:\n{ctx}\nRecent history: {history_snip}\n"
            "Respond ONLY with JSON. Example: {\"action\":\"call_ai\",\"action_input\":{...}}"
        )

    def _extract_json(self, text:str) -> Dict[str,Any]:
        first = text.find("{")
        last = text.rfind("}")
        if first == -1 or last == -1:
            raise ValueError("No JSON found")
        return json.loads(text[first:last+1])

    def _validate_action(self, obj:Dict[str,Any]) -> str:
        action = obj.get("action")
        if not action:
            raise ValueError("Missing action")
        if action != "finish" and action not in TOOLS:
            raise ValueError(f"Unknown action: {action}")
        return action

    def _execute_tool(self, action:str, action_input:Dict[str,Any]) -> Dict[str,Any]:
        if action == "finish":
            return {"done": True, "result": action_input}
        if action in self.approval_required:
            return {"need_approval": True, "action": action, "input": action_input}
        fn = TOOLS.get(action)
        if not fn:
            return {"error": f"tool {action} not implemented"}
        try:
            out = fn(action_input or {})
            return {"done": False, "observation": out}
        except Exception as e:
            logger.exception("Tool execution error")
            return {"done": False, "error": str(e)}

    def run_task(self, task:str, context_query:str=None) -> Dict[str,Any]:
        context = []
        if context_query and self.memory:
            try:
                context = self.memory.retrieve(context_query, top_k=5)
            except Exception:
                context = []
        history = []
        prompt = self._build_prompt(task, context, history)

        for step in range(self.max_steps):
            # choose reasoner model
            try:
                model = self.ai.recommend_platform(task)
            except Exception:
                model = None

            # call reasoner
            if not hasattr(self.ai, "call_reasoner"):
                raise RuntimeError("AiController.call_reasoner required")
            raw = self.ai.call_reasoner(model, prompt, temperature=0.0, max_tokens=800)
            try:
                action_obj = self._extract_json(raw)
            except Exception as e:
                logger.exception("Failed parsing LLM output")
                return {"status":"parse_error","error":str(e),"raw":raw}

            try:
                self._validate_action(action_obj)
            except Exception as e:
                return {"status":"invalid_action","error":str(e),"raw": action_obj}

            exec_res = self._execute_tool(action_obj.get("action"), action_obj.get("action_input"))

            entry = {"step": step, "action": action_obj, "exec": exec_res, "ts": time.time()}
            self._write_audit(entry)
            history.append(entry)

            if exec_res.get("need_approval"):
                return {"status":"awaiting_approval","entry":entry}

            if exec_res.get("done"):
                # evaluate and store
                try:
                    self._post_task_reflection(task, history, exec_res.get("result"))
                except Exception:
                    logger.exception("post reflection failed")
                return {"status":"finished","result": exec_res.get("result"), "history": history}

            # update prompt with concise observation
            obs = exec_res.get("observation")
            obs_summary = str(obs)[:1000]
            prompt = self._build_prompt(task, context + [obs_summary], history)

        return {"status":"failed_max_steps","history": history}

    def _post_task_reflection(self, task:str, history:List[Dict[str,Any]], result:Any):
        # call judge model
        try:
            judge_model = self.ai.default_judge_model()
        except Exception:
            judge_model = None

        eval_prompt = (
            f"Task: {task}\nHistory: {json.dumps(history, default=str)}\nResult: {json.dumps(result, default=str)}\n"
            "Return JSON: {'success': bool, 'metrics': {}, 'next_actions': ['string tasks']}"
        )

        ev = {"success": False, "metrics": {}, "next_actions": []}
        if judge_model and hasattr(self.ai, "call_reasoner"):
            try:
                out = self.ai.call_reasoner(judge_model, eval_prompt, temperature=0.0, max_tokens=400)
                ev = self._extract_json(out)
            except Exception:
                logger.exception("Judge parsing failed")

        # store evaluation
        try:
            self.memory.store({"task":task, "evaluation":ev, "history":history})
        except Exception:
            logger.exception("Memory store failed")

        # enqueue training tasks if any
        if not ev.get("success") and ev.get("next_actions"):
            for t in ev.get("next_actions", []):
                try:
                    self.training.enqueue_training_task(t)
                except Exception:
                    logger.exception("enqueue failed")

# ReAct Agent & Tools Bundle for Synapse Chamber
# ==============================================
# This is the foundation for your agentic AI system. It merges reasoning (ReAct framework),
# tool use, and self-improvement into a single orchestrated module.

from flask import Flask, request, jsonify
import openai
import json
import os
import random
import time

# Load environment variables
openai.api_key = os.getenv("OPENAI_API_KEY", "demo-key")

# ----------------------------
# 1. MEMORY SYSTEM
# ----------------------------
class MemorySystem:
    def __init__(self, path="memory.json"):
        self.path = path
        self.memory = self._load()

    def _load(self):
        if os.path.exists(self.path):
            with open(self.path, 'r') as f:
                return json.load(f)
        return {"sessions": []}

    def save(self):
        with open(self.path, 'w') as f:
            json.dump(self.memory, f, indent=2)

    def store(self, role, content):
        self.memory["sessions"].append({"role": role, "content": content})
        self.save()

    def retrieve_context(self, limit=5):
        return self.memory["sessions"][-limit:]


# ----------------------------
# 2. TOOLKIT (Action Layer)
# ----------------------------
class Toolset:
    def __init__(self):
        self.tools = {
            "search": self.search_web,
            "summarize": self.summarize_text,
            "calculate": self.calculate_expression
        }

    def search_web(self, query):
        return f"[Search Result Placeholder for: {query}]"

    def summarize_text(self, text):
        words = text.split()
        return ' '.join(words[:min(50, len(words))]) + ('...' if len(words) > 50 else '')

    def calculate_expression(self, expr):
        try:
            return eval(expr, {"__builtins__": {}})
        except Exception as e:
            return str(e)

    def use_tool(self, tool_name, *args):
        if tool_name in self.tools:
            return self.tools[tool_name](*args)
        return f"Tool '{tool_name}' not found."


# ----------------------------
# 3. REACT AGENT CORE
# ----------------------------
class ReActAgent:
    def __init__(self, memory: MemorySystem, tools: Toolset):
        self.memory = memory
        self.tools = tools
        self.name = "AutoDev ReAct"

    def think(self, prompt):
        context = '\n'.join([f"{x['role']}: {x['content']}" for x in self.memory.retrieve_context()])
        full_prompt = f"""You are {self.name}, a self-improving AI agent.
Context:
{context}

Task: {prompt}
Reason step-by-step using the ReAct (Reason + Act) framework.
If an action is required, use the format:
Action: <tool_name>[arguments]
Then reflect and give a final answer as:
Observation: <result of the action>
Final Answer: <your reflection or result>
"""
        response = openai.ChatCompletion.create(
            model="gpt-4-turbo",
            messages=[{"role": "system", "content": "You are a reasoning AI agent."},
                     {"role": "user", "content": full_prompt}],
            temperature=0.6
        )
        text = response.choices[0].message.content
        self.memory.store("assistant", text)
        return text

    def act(self, reasoning_output):
        if "Action:" in reasoning_output:
            try:
                action_line = reasoning_output.split("Action:")[1].split("\n")[0].strip()
                tool_name, arg = action_line.split("[", 1)
                arg = arg.rstrip("]")
                result = self.tools.use_tool(tool_name.strip(), arg)
                self.memory.store("tool_result", result)
                return f"Observation: {result}"
            except Exception as e:
                return f"Action parsing failed: {e}"
        return "No action required."


# ----------------------------
# 4. FLASK SERVER (Interface)
# ----------------------------
app = Flask(__name__)

memory = MemorySystem()
tools = Toolset()
agent = ReActAgent(memory, tools)

@app.route('/prompt', methods=['POST'])
def prompt_agent():
    data = request.get_json()
    user_input = data.get('prompt', '')
    memory.store('user', user_input)

    reasoning_output = agent.think(user_input)
    action_result = agent.act(reasoning_output)

    return jsonify({
        "reasoning": reasoning_output,
        "action_result": action_result
    })


@app.route('/')
def home():
    return "<h2>AutoDev ReAct Agent is Running 🧠⚙️</h2>"


if __name__ == '__main__':
    print("[BOOT] AutoDev ReAct Agent is initializing...")
    app.run(host='0.0.0.0', port=5000, debug=True)
